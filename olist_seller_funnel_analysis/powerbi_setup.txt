# STEP 2: DAX MEASURES TABLE
# Create a new blank table called "Measures" and add these measures
# ──────────────────────────────────────────────────────────────────────

# ── KPI MEASURES ──────────────────────────────────────────────────────

Total MQLs =
COUNTROWS('olist_master_dataset')

Won Deals =
CALCULATE(
    COUNTROWS('olist_master_dataset'),
    'olist_master_dataset'[converted] = 1
)

Overall Conversion Rate =
DIVIDE([Won Deals], [Total MQLs], 0)

Overall Conversion Rate % =
FORMAT([Overall Conversion Rate], "0.0%")

Avg Days to Close =
CALCULATE(
    AVERAGE('olist_master_dataset'[days_to_close]),
    'olist_master_dataset'[converted] = 1
)

Funnel Leak % =
1 - [Overall Conversion Rate]

Funnel Leak % Label =
FORMAT([Funnel Leak %], "0.0%") & " drop-off at MQL→SDR"


# ── CHANNEL MEASURES ──────────────────────────────────────────────────

Leads by Channel =
CALCULATE(
    [Total MQLs],
    ALLEXCEPT('olist_master_dataset', 'olist_master_dataset'[origin])
)

Deals by Channel =
CALCULATE(
    [Won Deals],
    ALLEXCEPT('olist_master_dataset', 'olist_master_dataset'[origin])
)

Channel Conversion Rate =
DIVIDE(
    CALCULATE([Won Deals], ALLEXCEPT('olist_master_dataset','olist_master_dataset'[origin])),
    CALCULATE([Total MQLs], ALLEXCEPT('olist_master_dataset','olist_master_dataset'[origin])),
    0
)

Channel Conversion Rate % =
FORMAT([Channel Conversion Rate], "0.00%")

Channel Quality Label =
SWITCH(
    TRUE(),
    [Channel Conversion Rate] >= 0.10, "Strong",
    [Channel Conversion Rate] >= 0.06, "Moderate",
    "Weak"
)


# ── SEGMENT MEASURES ──────────────────────────────────────────────────

Segment Avg Days =
CALCULATE(
    AVERAGE('olist_master_dataset'[days_to_close]),
    'olist_master_dataset'[converted] = 1,
    ALLEXCEPT('olist_master_dataset','olist_master_dataset'[business_segment])
)

Segment Deal Count =
CALCULATE(
    [Won Deals],
    ALLEXCEPT('olist_master_dataset','olist_master_dataset'[business_segment])
)

Segment Speed Label =
SWITCH(
    TRUE(),
    [Segment Avg Days] <= 35, "Fast (≤35d)",
    [Segment Avg Days] <= 55, "Moderate (36–55d)",
    "Slow (>55d)"
)


# ── MONTHLY TREND ─────────────────────────────────────────────────────

Monthly Leads =
CALCULATE(
    [Total MQLs],
    ALLEXCEPT('olist_master_dataset','olist_master_dataset'[month])
)

MoM Growth % =
VAR CurrentMonth = CALCULATE([Total MQLs])
VAR PrevMonth =
    CALCULATE(
        [Total MQLs],
        DATEADD('olist_master_dataset'[first_contact_date], -1, MONTH)
    )
RETURN
DIVIDE(CurrentMonth - PrevMonth, PrevMonth, BLANK())
